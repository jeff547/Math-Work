\noindent \textbf{Problem 3:}

Let $C_1 = $ event that the two-headed coin is picked,
$C_2= $ event that the fair coin is picked,
and $C_3= $ event the biased coin is picked,
finally, let $H =$ event that heads is flipped.

Thus, by using Bayes theorem, we have:
\begin{align*}
    P(C_1|H) & = \frac{P(H|C_1)*P(C_1)}{P(H|C_1)*P(C_1) + P(H|C_2)*P(C_2) + P(H|C_3)*P(C_3)}                \\
             & = \frac{\frac{1}{3}}{\frac{1}{3} + (\frac{1}{2}) (\frac{1}{3}) + (\frac{1}{3})(\frac{3}{4})} \\
             & = \tfrac{4}{9}
\end{align*}

\noindent \textbf{Problem 5:}

We have P(dog) = .36, P(cat) = .30, and P(cat $|$ dog) = 0.22
\begin{enumerate}
    \item[(a)]
          \begin{align*}
              \text{P(cat)} \cap{} \text{P(dog)} & = \text{P(cat $|$ dog)} * \text{P(dog)} \\
                                                 & = 0.22 * 0.36                           \\
                                                 & = 0.0792                                \\
                                                 & = 7.92\%
          \end{align*}
    \item[(b)]
          \begin{align*}
              \text{P(dog $|$ cat)} & = \frac{\text{P(dog)} \cap{} \text{P(cat)}}{\text{P(cat)}} \\
                                    & = \frac{0.0792}{0.3}                                       \\
                                    & = 0.264                                                    \\
                                    & = 26.4\%
          \end{align*}

\end{enumerate}
