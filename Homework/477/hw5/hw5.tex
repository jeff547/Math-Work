\noindent \textbf{Problem 1:}

Let X = Poisson(0.2), the number of typographical errors
\begin{itemize}
    \item [(a)]
          \begin{align*}
              P(X=0) = e^{-0.2} \approx{0.819}
          \end{align*}
    \item [(b)]
          \begin{align*}
              P(X\geq{2}) & = 1 - P(X=0) - P(X=1)   \\

                          & \approx{0.018}
          \end{align*}
\end{itemize}

\noindent \textbf{Problem 2:}

We have that $P(On) = \frac{1}{3}$ and $P(Off) = \frac{2}{3}$

\begin{itemize}
    \item Case A: 3 examiners
          \begin{align*}
              P(Pass) & = P(Off) (\binom{3}{2}(0.4)^2(0.6) + \binom{3}{3}(0.4)^3) + P(On) (\binom{3}{2}(0.8)^2(0.2) + \binom{3}{3}(0.8)^3) \\
                      & = \frac{2}{3} (0.352) + \frac{1}{3} (0.896)                                                                        \\
                      & \approx{0.533}
          \end{align*}

    \item Case B: 5 examiners
          \begin{align*}
              P(Pass | On)  & = \binom{5}{3}(0.8)^3(0.2)^2 + \binom{5}{4}(0.8)^4(0.2) + \binom{5}{5}(0.8)^5  \approx {0.942} \\
              P(Pass | Off) & = \binom{5}{3}(0.4)^3(0.6)^2 + \binom{5}{4}(0.4)^4(0.6) + \binom{5}{5}(0.4)^5 \approx {0.317}  \\
              P(Pass)       & = P(Pass | On) P(On) + P(Pass | Off) P(Off)                                                    \\
                            & \approx{0.525}
          \end{align*}
\end{itemize}

He will have a higher chance of passing with 3 examiners.
